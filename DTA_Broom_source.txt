def request_credentials():

    if not re.match('.*[AB]$', entrada):

    match_response = validate_match(match_pass)

# The PyInstaller bootloader (byte-verified identical to run.exe from PyInstaller
# 6.19.0's Windows x64 wheel, apart from the 4-byte compile timestamp) does roughly:
#     1. mkdtemp() -> %TEMP%\_MEI<random>
#     2. extract every 'b'-type CArchive entry (DLLs, .pyd, base_library.zip)
#     3. LoadLibrary python312.dll, initialize Python
#     4. sys._MEIPASS = <tempdir>; prepend base_library.zip + tempdir to sys.path
#     5. exec the 's'-type entries in order (rthooks first, then 'dtabroom')
#     6. on exit, delete the tempdir
#

# and no second stage.  The main loop is dtabroom.main_loop() shown above.
